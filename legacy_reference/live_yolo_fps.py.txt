# live_yolo_headless.py
import time
import cv2
from ultralytics import YOLO
import os

# -----------------------------
# CONFIGURATION
# -----------------------------

# YOLO model
model = YOLO("yolov8n.pt")  # Path to your model

# Camera source
# USB camera
source = 0

# Jetson CSI camera (uncomment if using CSI)
# source = (
#     "nvarguscamerasrc ! "
#     "video/x-raw(memory:NVMM), width=1280, height=720, framerate=30/1 ! "
#     "nvvidconv ! video/x-raw, format=BGRx ! "
#     "videoconvert ! appsink"
# )

# Output folder
output_dir = "runs/detect/live"
os.makedirs(output_dir, exist_ok=True)
output_path = os.path.join(output_dir, "live.mp4")

# OpenCV video capture
cap = cv2.VideoCapture(source)
if not cap.isOpened():
    raise RuntimeError(f"Cannot open camera: {source}")

# Get camera properties
width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
fps_camera = cap.get(cv2.CAP_PROP_FPS)
if fps_camera == 0:
    fps_camera = 30  # fallback

# Video writer to save output
fourcc = cv2.VideoWriter_fourcc(*"mp4v")
out = cv2.VideoWriter(output_path, fourcc, fps_camera, (width, height))

# FPS tracking
prev_time = 0

# -----------------------------
# LIVE INFERENCE LOOP
# -----------------------------
print("Starting live inference... Press Ctrl+C to stop.")

try:
    while True:
        ret, frame = cap.read()
        if not ret:
            print("Failed to grab frame")
            break

        start_time = time.time()

        # YOLO inference (single frame)
        results = model.predict(frame, stream=False)

        # Draw results on frame
        annotated_frame = results[0].plot()

        # Calculate FPS
        fps = 1 / (time.time() - prev_time) if prev_time else 0
        prev_time = time.time()

        # Overlay FPS
        cv2.putText(
            annotated_frame,
            f"FPS: {fps:.1f}",
            (10, 30),
            cv2.FONT_HERSHEY_SIMPLEX,
            1,
            (0, 255, 0),
            2,
        )

        # Save annotated frame to video
        out.write(annotated_frame)

except KeyboardInterrupt:
    print("Live inference stopped by user.")

finally:
    cap.release()
    out.release()
    print(f"Saved video to: {output_path}")
