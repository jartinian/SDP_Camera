import serial
import time

import cv2
from flask import Flask, Response
from ultralytics import YOLO

# --------------------
# YOLO + CAMERA
# --------------------
model = YOLO("yolov8n.pt")
cap = cv2.VideoCapture(0)

###
uart = serial.Serial(
    "/dev/ttyTHS1",   # Jetson UART
    baudrate=115200,
    timeout=0         # non-blocking
)

time.sleep(2)
###

if not cap.isOpened():
    raise RuntimeError("Could not open camera")

app = Flask(__name__)

def generate_frames():
    while True:
        ret, frame = cap.read()
        if not ret:
            break

###
        results = model.predict(frame, stream=False)
        r = results[0]

        if r.boxes is not None:
            for box in r.boxes:
                # Integer bounding box
                x_min, y_min, x_max, y_max = map(int, box.xyxy[0].tolist())
                class_id = int(box.cls[0])
                conf = int(float(box.conf[0]) * 100)

                # UART message (ALL INTEGERS)
                message = f"{class_id},{x_min},{y_min},{x_max},{y_max},{conf}\n"
                uart.write(message.encode("ascii"))

                # Non-blocking ACK read
                if uart.in_waiting:
                    ack = uart.readline().decode("ascii").strip()
                    if ack == "ACK":
                        pass  # optional logging

                break  # send only ONE detection per frame
###		
        annotated = results[0].plot()

        _, buffer = cv2.imencode(".jpg", annotated)
        frame_bytes = buffer.tobytes()

        yield (b"--frame\r\n"
               b"Content-Type: image/jpeg\r\n\r\n" +
               frame_bytes + b"\r\n")

@app.route("/")
def index():
    return """
    <html>
        <head><title>YOLO Live Feed</title></head>
        <body>
            <h1>YOLOv8 Live Feed</h1>
            <img src="/video">
        </body>
    </html>
    """

@app.route("/video")
def video():
    return Response(generate_frames(),
                    mimetype="multipart/x-mixed-replace; boundary=frame")

if __name__ == "__main__":
    print("Starting YOLO stream on http://0.0.0.0:8080")
    app.run(host="0.0.0.0", port=8080, threaded=True)
