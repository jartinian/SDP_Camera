import time
import cv2
from ultralytics import YOLO
import os
from flask import Flask, Response
import threading

app = Flask(__name__)
output_frame = None
lock = threading.Lock()

def generate_frames():
    global output_frame, lock

    while True:
        with lock:
            if output_frame is None:
                continue

            ret, buffer = cv2.imencode(".jpg", output_frame)
            if not ret:
                continue

            frame = buffer.tobytes()

        yield (b"--frame\r\n"
               b"Content-Type: image/jpeg\r\n\r\n" + frame + b"\r\n")

@app.route("/")
def index():
    return """
    <html>
        <head><title>YOLO Live Feed</title></head>
        <body>
            <h1>YOLOv8 Live Feed</h1>
            <img src="/video">
        </body>
    </html>
    """

@app.route("/video")
def video():
    return Response(generate_frames(),
                    mimetype="multipart/x-mixed-replace; boundary=frame")

def yolo_loop():
    global output_frame, lock

    print("Starting live inference...")

    while True:
        ret, frame = cap.read()
        if not ret:
            print("Failed to grab frame")
            break

        start_time = time.time()


        
        #############
        results = model(frame)

        for r in results:
            if r.boxes is None:
                continue

            for box in r.boxes:
                # Bounding box (pixels → int)
                x_min, y_min, x_max, y_max = box.xyxy[0].tolist()
                x_min = int(x_min)
                y_min = int(y_min)
                x_max = int(x_max)
                y_max = int(y_max)

                # Class ID (already integer, but be explicit)
                class_id = int(box.cls[0])

                # Confidence scaled to integer (0–100)
                conf_int = int(float(box.conf[0]) * 100)
                
                # Angle init
                center_angle = 0

                # UART message (all integers)
                message = f"{class_id},{x_min},{y_min},{x_max},{y_max},{conf_int}\n"
                

                ##########
                center_x = 0
                if class_id == 11:
                    center_x = (x_min + x_max) / 2
                    center_angle = (center_x - 320) * (78 / 640)
                    message = ""

                # RIGHT
                    if center_x > 330:
                        message += "R"
                    else:
                        message += "-"

                # LEFT
                    if center_x < 310:
                        message += "L"
                    else:
                        message += "-"

                # CENTER
                    if 310 <= center_x <= 330:
                        message += "C"
                    else:
                        message += "-"

                # UP / CLOSE
                    if y_min < 50:
                        message += "U"
                    else:
                        message += "-"

                    message += "\n"

                #uart.write(message.encode("ascii"))
                message1 = f"{class_id},{x_min},{y_min},{x_max},{y_max},{conf_int},{center_angle}\n"
                
                ##########

                current_time = time.time()

                if current_time - last_uart_time >= UART_INTERVAL:
                    def yolo_loop():

                    uart.write(message1.encode("ascii"))

                if uart.in_waiting > 0:
                    ack = uart.readline().decode("ascii").strip()
                    if ack == "ACK":
                        print("ACK received")
        ###############


        # Draw results on frame
        annotated_frame = results[0].plot()
